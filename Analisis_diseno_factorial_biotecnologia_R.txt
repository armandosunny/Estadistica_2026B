
############################################################
# DISEÑO FACTORIAL 3 x 2 EN BIOTECNOLOGÍA
# Ejemplo: expresión de proteína recombinante soluble en E. coli
#
# Factor A: temperatura de inducción = 20, 30, 37 °C
# Factor B: IPTG = 0.1, 0.5 mM
# Réplicas: 5 por combinación
# Variable respuesta: proteina_soluble_mg_L
#
# OBJETIVOS:
# 1) Evaluar el efecto de la temperatura.
# 2) Evaluar el efecto de la concentración de IPTG.
# 3) Evaluar la interacción temperatura x IPTG.
# 4) Revisar supuestos del ANOVA factorial.
# 5) Realizar comparaciones post hoc / efectos simples.
# 6) Obtener tamaños de efecto y gráficas de publicación.
############################################################

datos <- datos %>%

# 4. ESTADÍSTICA DESCRIPTIVA
# ============================================================

resumen <- datos %>%
  group_by(temperatura, IPTG) %>%
  summarise(
    n = n(),
    media = mean(proteina_soluble_mg_L),
    DE = sd(proteina_soluble_mg_L),

    mediana = median(proteina_soluble_mg_L),
    minimo = min(proteina_soluble_mg_L),
    maximo = max(proteina_soluble_mg_L),
    .groups = "drop"
  )

# ============================================================
# 6. MODELO FACTORIAL COMPLETO
# ============================================================

modelo <- lm(
  proteina_soluble_mg_L ~ temperatura * IPTG,
  data = datos
)

summary(modelo)

# ANOVA factorial:
anova(modelo)

# También se puede usar:
modelo_aov <- aov(
  proteina_soluble_mg_L ~ temperatura * IPTG,
  data = datos
)

summary(modelo_aov)


# ============================================================
# 7. ANOVA TIPO II Y TIPO III
# ============================================================

# Para un diseño BALANCEADO como éste, el ANOVA clásico es suficiente.
# Se muestran también estas variantes con fines didácticos.

Anova(modelo, type = 2)

# Tipo III:
# Es importante usar contrastes de suma antes de ajustar el modelo.

options(contrasts = c("contr.sum", "contr.poly"))

modelo_tipo3 <- lm(

# ============================================================
# INTERPRETACIÓN DE emmeans Y COMPARACIONES POR TEMPERATURA
# ============================================================

# emmeans calcula las medias marginales estimadas de cada nivel
# de IPTG dentro de cada temperatura.
#
# Esto permite estudiar los EFECTOS SIMPLES del IPTG, es decir:
# evaluar si el efecto del IPTG cambia dependiendo de la temperatura.
#
# A 20 °C:
# 0.5 mM produce más proteína que 0.1 mM
# diferencia ≈ 10.3 mg/L, p < 0.0001
#
# A 30 °C:
# 0.5 mM produce más proteína que 0.1 mM
# diferencia ≈ 20.2 mg/L, p < 0.0001
#
# A 37 °C:
# ocurre lo contrario:
# 0.1 mM produce más proteína que 0.5 mM
# diferencia ≈ 5.7 mg/L, p < 0.0001
#
# CONCLUSIÓN:
# El efecto del IPTG depende de la temperatura.
# A 20 y 30 °C, aumentar IPTG favorece la producción de proteína,
# pero a 37 °C aumentar IPTG reduce la producción.
#
# Este cambio en la magnitud e incluso en la dirección del efecto
# es una evidencia clara de interacción temperatura × IPTG.

# ============================================================
# INTERPRETACIÓN DE LA GRÁFICA DE P-VALORES DE TUKEY
# ============================================================

# Cada panel corresponde a una concentración de IPTG.

# Los puntos muestran las medias marginales estimadas
# de producción de proteína para cada temperatura.

# Las líneas conectan las temperaturas que se comparan entre sí.

# El eje horizontal muestra el p-valor ajustado por Tukey.

# p < 0.05:
# las dos temperaturas presentan una diferencia significativa.

# En este ejercicio, las comparaciones entre temperaturas
# son significativas dentro de ambos niveles de IPTG.

# Con IPTG = 0.1 mM:
# 30 °C > 37 °C > 20 °C.

# Con IPTG = 0.5 mM:
# 30 °C > 20 °C > 37 °C.

# El cambio en el orden de las temperaturas entre los dos niveles
# de IPTG evidencia que el efecto de la temperatura depende
# de la concentración de IPTG, es decir, existe interacción.

# ============================================================
# MATRIZ DE COMPARACIONES TUKEY
# ============================================================

tukey_todas <- pairs(
  emm_combinaciones,
  adjust = "tukey"
)

tukey_df <- as.data.frame(tukey_todas)

tukey_df




library(tidyr)

tukey_heat <- tukey_df %>%
  separate(
    contrast,
    into = c("Tratamiento_1", "Tratamiento_2"),
    sep = " - "
  )



tukey_heat <- tukey_heat %>%
  mutate(
    significancia = case_when(
      p.value < 0.001 ~ "p < 0.001",
      p.value < 0.01  ~ "p < 0.01",
      p.value < 0.05  ~ "p < 0.05",
      TRUE            ~ "NS"
    )
  )



ggplot(
  tukey_heat,
  aes(
    x = Tratamiento_1,
    y = Tratamiento_2,
    fill = p.value
  )
) +

  geom_tile(
    color = "white",
    linewidth = 1.5
  ) +

  geom_text(
    aes(label = significancia),
    size = 4,
    fontface = "bold"
  ) +

  scale_fill_gradient(
    low = "darkred",
    high = "white",
    limits = c(0, 1)
  ) +

  labs(
    title = "Comparaciones múltiples de Tukey",
    subtitle = "Temperatura × concentración de IPTG",
    x = NULL,
    y = NULL,
    fill = "p-valor"
  ) +

  theme_minimal(base_size = 13) +

  theme(
    panel.grid = element_blank(),

    axis.text.x = element_text(
      angle = 45,
      hjust = 1
    ),

    plot.title = element_text(
      face = "bold",
      hjust = 0.5
    ),

    plot.subtitle = element_text(
      hjust = 0.5
    )
  )
