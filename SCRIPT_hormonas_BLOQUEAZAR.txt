# Cambia la ruta si es necesario
datos <- read_excel(

  sheet = "Datos"
)

# Ver datos

# ============================================================
# 2. PREPARAR VARIABLES
# ============================================================

datos <- datos %>%
  mutate(
    bloque = factor(
      bloque,
      levels = paste0("B", 1:10)
    ),

    concentracion = factor(
      concentracion_ng_mL,
      levels = c(0, 25, 50, 100, 200)
    ),

    orden_en_bloque = as.numeric(orden_en_bloque),

    tiempo_min = as.numeric(tiempo_min),

    OD450 = as.numeric(OD450)
  )


# Revisar
str(datos)

head(datos)

#MODELO DBCA

m1 <- lm(
  OD450 ~ bloque + concentracion,
  data = datos
)

anova(m1)

summary(m1)

#Revisar supuestos del ANOVA



diagnostico <- datos %>%
  mutate(
    ajustado = fitted(m1),
    residuo = residuals(m1),
    residuo_student = rstudent(m1),
    cook = cooks.distance(m1)
  )

head(diagnostico)


#Normalidad de residuos: histograma

p_hist <- ggplot(
  diagnostico,
  aes(x = residuo)
) +

  geom_histogram(
    bins = 10,
    color = "white",
    alpha = 0.8
  ) +

  geom_vline(
    xintercept = 0,
    linetype = 2

  labs(

    x = "Residuo",
    y = "Frecuencia"
  ) +

  theme_classic(base_size = 14)

p_hist
